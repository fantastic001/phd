\section{Motivation}
% ============================================================

\begin{frame}{Motivation}
  \begin{itemize}
    \item Large graphs that do not fit on one machine are embedded by partitioning the graph, embedding each partition independently, and merging the results.
    \item Each partition has its own skip-gram embedding space, so vectors from different partitions are not directly comparable.
    \item Link prediction asks whether embeddings generalize to edges that were \textbf{not} seen during training, which is what downstream recommendation depends on.
    \item Seminar 3 studied reconstruction only. Does a good reconstruction score imply good link prediction?
  \end{itemize}
\end{frame}

\begin{frame}{Problem Formulation}
  Given a static graph partitioned into $P$ partitions, each embedded independently with static node2vec, how do
  \begin{itemize}
    \item the partition count $P$ and the partitioning strategy (LFM vs.\ LPA),
    \item the link-prediction decoder (bilinear, Hadamard, random forest)
  \end{itemize}
  affect link-prediction quality, and how does that quality relate to reconstruction quality?
\end{frame}

% ============================================================
\section{Method}
% ============================================================

\begin{frame}{Partitioned Embedding Pipeline}
  \begin{enumerate}
    \item Hold out $10\%$ of the edges as positive test pairs; draw an equal number of non-edges as negatives.
    \item Partition the training graph once into $P$ partitions.
    \item Embed each partition's induced subgraph independently with static node2vec.
    \item Merge per-partition embeddings into one embedding per vertex; train the decoder on the merged embeddings.
    \item Evaluate the decoder on the held-out pairs.
  \end{enumerate}
  \vspace{0.5em}
  A pair whose endpoints lie in different partitions is scored from two independently trained spaces; this is expected to drive most of the loss.
\end{frame}

\begin{frame}{Partitioners}
  Both are community-detection based: detect communities, then pack them into exactly $P$ partitions with a size-balancing greedy bin-packing step.
  \begin{itemize}
    \item \textbf{LPA} (label propagation): communities are disjoint, so every vertex belongs to exactly one partition. No tunable parameters.
    \item \textbf{LFM} (Lancichinetti--Fortunato--Kertesz): greedy fitness-based growth from seed vertices; communities may overlap, so a vertex can appear in several partitions.
  \end{itemize}
  \vspace{0.5em}
  All main experiments use LPA.
\end{frame}

\begin{frame}{Link-Prediction Decoders}
  Given merged embeddings $u = z(a)$, $v = z(b)$:
  \begin{itemize}
    \item \textbf{Hadamard}: $s(u,v) = w^\top (u \odot v) = \sum_{i} w_i u_i v_i$
    \item \textbf{Bilinear}: $s(u,v) = u^\top W v$, with $W \in \mathbb{R}^{d \times d}$ unrestricted
    \item \textbf{Random forest}: 20 trees on the concatenated vector $[u; v] \in \mathbb{R}^{2d}$, axis-aligned splits on raw coordinates
  \end{itemize}
  \vspace{0.5em}
  Neural decoders: $\hat{y} = \sigma(s(u,v))$, binary cross-entropy, minibatch SGD. An edge is predicted when $\hat{y} > 0.5$.
\end{frame}

\begin{frame}{Decoder Ordering}
  \begin{itemize}
    \item Most configurations are ordered \textbf{random forest $>$ bilinear $>$ Hadamard}.

    \item Bilinear beats Hadamard significantly in 14 of 16 groups on MRR.
    \item Random forest beats bilinear on MRR in 14 of 16 groups; its advantage grows with $P$ (AS-Oregon: $d$ from $+14.5$ to $+21.2$).

  \end{itemize}

\begin{frame}{Reconstruction vs.\ Link Prediction}
  \begin{itemize}
    \item Pooled Spearman $\rho$ between per-run F1 and MRR is positive for all decoders ($\rho = 0.62$--$0.72$, $n=160$).
    \item This is driven by dataset differences: graphs that are easy to embed score high on both.
    \item Within a dataset, increasing $P$ pushes the two quantities apart.
  \end{itemize}
  \vspace{0.5em}
  Reconstruction F1 indicates how learnable a graph is, but is \textbf{not} a reliable proxy for how well partitioned embeddings support link prediction.
\end{frame}

\section{Conclusion}
% ============================================================

\begin{frame}{Conclusion}
  \begin{itemize}
    \item Partitioning strongly degrades neural decoders: bilinear MRR falls by 39--60\% and Hadamard by 8--39\% from $P=1$ to $P=8$.
    \item Random forest on concatenated embeddings is far less sensitive ($+4\%$ to $-26\%$).
    \item Decoder ordering: random forest $>$ bilinear $>$ Hadamard in most configurations; bilinear is best only on unpartitioned dense graphs.
    \item Reconstruction F1 rises with $P$ while link prediction falls, so reconstruction is not a proxy for downstream link prediction.
    \item LPA beats LFM in a preliminary single-dataset comparison.
  \end{itemize}
\end{frame}

\begin{frame}{Future Work}
  \begin{itemize}
    \item Split test pairs into within-partition and cross-partition pairs to test whether cross-partition pairs cause the degradation.
    \item Align per-partition embedding spaces (e.g.\ with a learned linear map) before decoding.
    \item Tune the neural decoders per partition count; repeat ranking evaluation with random tie-breaking.
    \item Extend the LFM vs.\ LPA comparison and the decoder study to more datasets.
    \item Study the replication factor as a way to recover link-prediction quality.
  \end{itemize}
\end{frame}
